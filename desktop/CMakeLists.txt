cmake_minimum_required(VERSION 3.21)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../VERSION" APP_VERSION)
string(STRIP "${APP_VERSION}" APP_VERSION)
project(SpearHeadDesktop VERSION ${APP_VERSION} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets Network Qml Quick QuickControls2 Svg Test DBus)
set(SOURCES bridge.cpp bridge.h nativeicon.cpp nativeicon.h api_client.cpp api_client.h)
file(GLOB QML_ASSETS CONFIGURE_DEPENDS qml/*.qml qml/qmldir assets/*.json assets/*.svg assets/*.png)
add_executable(spearhead-desktop tray.cpp api_client.cpp api_client.h)
target_link_libraries(spearhead-desktop PRIVATE Qt6::Widgets Qt6::Network)
target_compile_definitions(spearhead-desktop PRIVATE SPEARHEAD_VERSION="${PROJECT_VERSION}")
add_executable(spearhead-window window.cpp ${SOURCES})
target_compile_definitions(spearhead-window PRIVATE SPEARHEAD_VERSION="${PROJECT_VERSION}")
qt_add_resources(spearhead-window "ui" PREFIX "/ui" BASE "${CMAKE_CURRENT_SOURCE_DIR}" FILES ${QML_ASSETS})
target_link_libraries(spearhead-window PRIVATE Qt6::Widgets Qt6::Network Qt6::Quick Qt6::Qml Qt6::QuickControls2 Qt6::Svg Qt6::DBus)
include(CTest)
if(BUILD_TESTING)
  add_executable(spearhead-desktop-test desktop_test.cpp ${SOURCES})
  qt_add_resources(spearhead-desktop-test "ui" PREFIX "/ui" BASE "${CMAKE_CURRENT_SOURCE_DIR}" FILES ${QML_ASSETS})
  target_compile_definitions(spearhead-desktop-test PRIVATE SPEARHEAD_TEST_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}" SPEARHEAD_VERSION="${PROJECT_VERSION}")
  target_link_libraries(spearhead-desktop-test PRIVATE Qt6::Widgets Qt6::Network Qt6::Quick Qt6::Qml Qt6::QuickControls2 Qt6::Svg Qt6::Test Qt6::DBus)
endif()
